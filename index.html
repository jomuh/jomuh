    <!-- Live Calculation Engine Script -->

    <script>

        function calculateMargins() {

            // Fetch inputs

            const cupSize = parseFloat(document.getElementById('cupSize').value) || 0;

            const dailySales = parseFloat(document.getElementById('dailySales').value) || 0;

            const ingCostPerLitre = parseFloat(document.getElementById('ingCost').value) || 0;

            const pkgCostPerUnit = parseFloat(document.getElementById('pkgCost').value) || 0;

            const sellPrice = parseFloat(document.getElementById('sellPrice').value) || 0;

            const fixedCosts = parseFloat(document.getElementById('fixedCosts').value) || 0;

 

            // Operational Logic Metrics

            const monthlyVolumeUnits = dailySales * 30;

            const ingredientCostPerUnit = (cupSize / 1000) * ingCostPerLitre;

            const totalVariableCostPerUnit = ingredientCostPerUnit + pkgCostPerUnit;

            

            const monthlyRevenue = monthlyVolumeUnits * sellPrice;

            const monthlyVariableCosts = monthlyVolumeUnits * totalVariableCostPerUnit;

            const monthlyNetProfit = monthlyRevenue - monthlyVariableCosts - fixedCosts;

            

            const netMarginPercent = monthlyRevenue > 0 ? ((monthlyNetProfit / monthlyRevenue) * 100).toFixed(1) : 0;

 

            // Render calculations cleanly onto display badges

            document.getElementById('outTotalUnits').innerText = monthlyVolumeUnits.toLocaleString();

            document.getElementById('outCogs').innerText = 'KES ' + totalVariableCostPerUnit.toFixed(2);

            document.getElementById('outNetProfit').innerText = 'KES ' + monthlyNetProfit.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2});

            document.getElementById('outMargin').innerText = 'Net Margin: ' + netMarginPercent + '%';

 

            // Dynamic Risk Threshold Styling

            const banner = document.getElementById('profitBanner');

            if(monthlyNetProfit < 0) {

                banner.className = "p-4 rounded-lg bg-red-950 border border-red-700 text-center";

                banner.firstElementChild.className = "block text-xs uppercase font-bold text-red-400";

                document.getElementById('outNetProfit').className = "text-3xl font-black block my-1 text-red-500";

            } else {

                banner.className = "p-4 rounded-lg bg-gray-800 border border-green-600 text-center";

                banner.firstElementChild.className = "block text-xs uppercase font-bold text-green-400";

                document.getElementById('outNetProfit').className = "text-3xl font-black block my-1 text-green-400";

            }

        }

 

        // Attach recalculation event triggers to every field for realtime execution

        const inputs = ['cupSize', 'dailySales', 'ingCost', 'pkgCost', 'sellPrice', 'fixedCosts'];

        inputs.forEach(id => document.getElementById(id).addEventListener('input', calculateMargins));

        

        // Run baseline setup on first launch

        calculateMargins();

    </script>
